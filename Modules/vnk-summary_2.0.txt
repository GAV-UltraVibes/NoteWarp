    f.oblit=/облитерирован|облитерац|смыкани/i.test(t);
    f.phlebitis=f.masses&&(f.fresh||/верхушк|флотиру|флотаци/i.test(t));
    f.ptsSub=!f.phlebitis&&(f.recan||/тяжи|посттромбофлебитич|облитерирован|флеболит/i.test(t));
    var m=/флотаци[а-яё]*[^.]{0,25}?(\d+[.,]?\d*)\s*(см|мм)/i.exec(t)||/подвижн[а-яё]*[^.]{0,40}?(\d+[.,]?\d*)\s*(см|мм)/i.exec(t);
    f.float=m?m[1].replace(".",",")+" "+m[2].toLowerCase():null;
    f.nofloat=/без флотаци|флотаци[а-яё]* не выявлен|не флотиру|фиксирован[а-яё]* к стенк/i.test(t);
    var ma=/верхушк[а-яё]*[^.]{0,40}?((?:на уровне|в)\s+[^.]{4,60}?)(?:,|\.|$)/i.exec(t); f.apex=ma?ma[1].trim():null;
    f.spread=/распространен[а-яё]*|переходящ[а-яё]*|по перфоранту/i.test(t)&&/суральн/i.test(t);
    f.patent=/полностью сжимаем|прокрашива[а-яё]*\s+в\s+ЦДК/i.test(t);
    f.phlebitisWall=/стенки[^.]{0,30}(отёчн|утолщ|гипоэхоген)|перивенозн/i.test(t)&&!f.masses;
    return f; },
  subRender:function(f,supine,star){
    var out=[],g=VNK_SV[f.vein]?VNK_SV[f.vein].gen:"подкожной вены",s=f.side,C=this.CONFIG;
    var veinName=f.accessory?(f.accessory+" ствола "+g):(f.varix==="trib"&&f.drained?"дренируемого притока":g);
    if(f.postop){ var t="Состояние после "+f.postop+" "+(f.mainOnly?"основного ствола ":"")+g+" "+VNK_SIDE[s];
      if(f.stump) t+=", культя "+f.stump; out.push(t+"."); }
    if(f.phlebitis){ var tail=[];
      if(f.recan) tail.push(f.recan);
      if(f.apex) tail.push("верхушка "+f.apex); else if(C.placeholders) tail.push("верхушка "+star);
      if(f["float"]) tail.push("флотация "+f["float"]);
      else if(f.nofloat) tail.push("флотации не выявлено"+(supine&&C.supineNote?" (в положении лёжа)":""));
      out.push("Тромбофлебит "+(f.varix?"варикозно изменённой ":"")+veinName+" "+VNK_SIDE[s]+(tail.length?", "+tail.join(", "):"")+
        (f.spread?" с распространением на суральные вены":"")+"."); }
    if(f.ptsSub) out.push("Посттромбофлебитические изменения "+veinName+" "+VNK_SIDE[s]+", "+(f.recan||(f.oblit?"облитерация":"полная реканализация"))+".");
    if(f.phlebitisWall) out.push("Признаки флебита "+g+" "+VNK_SIDE[s]+".");
    var ins=[];
    if(f.sfj) ins.push("сафенофеморального соустья");
    if(f.spj) ins.push("сафенопоплитеального соустья");
    if(f.trunk&&!(C.subsumeTrunk&&(f.sfj||f.spj))) ins.push("клапанов ствола "+g);
    if(f.perf) ins.push("перфорантных вен"+(f.region?" "+f.region:""));
    var ch=[];
    if(ins.length) ch.push("несостоятельность "+ins.join(", "));
    if(f.varix==="trunk+trib") ch.push("варикозные изменения "+veinName+" с притоками");
    else if(f.varix==="trib") ch.push("варикозные изменения "+(f.drained?"дренируемых притоков ":"притоков ")+g);
    else if(f.varix==="trunk") ch.push("варикозные изменения "+veinName);
    if(ch.length) out.push(this.cap(ch.join(", "))+" "+VNK_SIDE[s]+".");
    return out; },

  /* ---------- разбор документа ---------- */
  conclusionRange:function(text){ var m=/^[ \t]*ЗАКЛЮЧЕНИЕ\s*:/im.exec(text); if(!m) return null;
    var st=/\n[ \t]*\n|\n[ \t]*Врач\s*:/i.exec(text.slice(m.index));
    return {start:m.index,end:st?m.index+st.index:text.length}; },
  analyze:function(text){
    var d=/^[ \t]*Врач\s*:/im.exec(text), body=d?text.slice(0,d.index):text;
    var r=this.conclusionRange(body); if(r) body=body.slice(0,r.start)+body.slice(r.end);
    var supine=/лёжа|клиностаз/i.test(body)||(!/в ортостазе/i.test(body)&&!/диаметр/i.test(body));
    var raw=body.replace(/\u00A0/g," "), facts=[], unknown=[], stops=[], scope=null, side=null, vein=null, self=this;
    raw.split(/\r?\n/).forEach(function(lineRaw){
      var line=lineRaw.replace(/\t/g," ").trim(); if(!line) return;
      if(/^(дуплексн|уздс|ультразвуков|цветовое|измерения\s+проводились|исследование затруднено)/i.test(line)){
        if(/затруднено/i.test(line)) stops.push(line); return; }
      var m;
      if((m=/^глубокие\s+вены\s*:?\s*(.*)$/i.exec(line))){ scope="deep"; side=null; vein=null; line=m[1]; }
      else if((m=/^подкожные\s+вены\s*:?\s*(.*)$/i.exec(line))){ scope="subcut"; side=null; vein=null; line=m[1]; }
      else if((m=/^(справа|слева)\s*:?\s*(.*)$/i.exec(line))){ side=m[1].toLowerCase()==="справа"?"right":"left"; vein=null; line=m[2]; }
      if(!line) return;
      self.sentences(self.normalize(line)).forEach(function(s){
        var t=s.replace(/\s+/g," ").trim(); if(!t) return;
        var sd=self.sideOf(t)||side; if(self.sideOf(t)) side=sd;
        if(VNK_STOP.test(t)){ stops.push(t); return; }
        if(scope==="deep"){
          if(/визуализированы общие|^остальные визуализированные/i.test(t)&&!/массы|тяжи|несост/i.test(t)){
            facts.push({code:"patent",scope:"deep",src:t}); return; }
          var f=self.deepFact(t,sd||"both"); f.code="deep"; facts.push(f);
          if(!f.masses&&!f.valves&&!f.outflow&&!f.compress&&!/визуализирован/i.test(t)) unknown.push(t);
        } else {
          var v=/перфорант/i.test(t)&&!VNK_SV.GSV.re.test(t)&&!VNK_SV.SSV.re.test(t)?null:
                (VNK_SV.GSV.re.test(t)?"GSV":VNK_SV.SSV.re.test(t)?"SSV":vein);
          if(v) vein=v;
          var sf=self.subFact(t,sd,v||"GSV"); sf.code="sub"; facts.push(sf);
          if(sf.patent) facts.push({code:"patent",scope:"subcut",src:t});
          if(!sf.postop&&!sf.varix&&!sf.perf&&sf.sfj===null&&sf.spj===null&&sf.trunk===null&&!sf.masses&&!sf.patent&&!sf.phlebitisWall)
            unknown.push(t);
        }
      });
    });
    return {facts:facts,unknown:unknown,stops:stops,supine:supine}; },

  /* ---------- сборка заключения ---------- */
  build:function(r){
    var self=this,C=this.CONFIG,star=(window.editor&&window.editor.jumpChar)||"*";
    var by={right:[],left:[],both:[]}, patent={}, noReflux=true;
    r.facts.forEach(function(f){
      var arr=f.code==="deep"?self.deepRender(f,r.supine,star):f.code==="sub"?self.subRender(f,r.supine,star):[];
      if(f.code==="patent"){ patent[f.scope]=true; return; }
      if(arr.length){ (by[f.side||"both"]=by[f.side||"both"]||[]).push({f:f,lines:arr}); if(/несостоятельн/i.test(arr.join(" "))) noReflux=false; }
    });
    var order=C.sideOrder.concat(["both"]).filter(function(s){ return by[s]&&by[s].length; });
    var multi=order.filter(function(s){ return s!=="both"; }).length>1;
    var rank=function(x){ return x.f.postop?0:x.f.phlebitis||x.f.code==="deep"?1:2; };
    var lines=[];
    order.forEach(function(s){
      var items=by[s].slice().sort(function(a,b){ return rank(a)-rank(b); });
      var txt=items.map(function(x){ return x.lines.join(" "); }).join(" ");
      lines.push(multi?((s==="both"?"С обеих сторон":self.cap(VNK_SIDE[s]))+": "+txt):txt);
    });
    if(noReflux&&!r.unknown.length&&!r.stops.length) lines.push("Клапанной недостаточности не выявлено.");
    var deepOK=patent.deep&&!r.stops.length, subOK=patent.subcut&&!r.stops.length;
    if(deepOK&&subOK) lines.push("Просветы глубоких и подкожных вен свободны.");
    else if(deepOK) lines.push("Просветы глубоких вен свободны.");
    if(!lines.length) return "";
    return multi?"ЗАКЛЮЧЕНИЕ:\n"+lines.join("\n"):"ЗАКЛЮЧЕНИЕ: "+lines.join(" "); },

  place:function(text,c){ var r=this.conclusionRange(text);
    if(r) return {text:text.slice(0,r.start)+c+text.slice(r.end),cursor:r.start+c.length};
    var d=/^[ \t]*Врач\s*:/im.exec(text);
    if(d) return {text:text.slice(0,d.index)+c+"\n\n"+text.slice(d.index),cursor:d.index+c.length};
    var tail=text.replace(/\s+$/,""); return {text:tail+"\n\n"+c+"\n",cursor:tail.length+2+c.length}; },
  run:function(api){
    var io=this.io(api),text=io.get();
    if(!/подкожн[а-яё]*\s+вен|БПВ|МПВ/i.test(text)){ io.hint("❌ Не похоже на протокол ВНК",2500); return; }
    var r=this.analyze(text); this.last=r;
    var c=this.build(r);
    if(!c){ io.hint("⚠️ Не удалось собрать заключение — см. «Разбор ВНК»",5000); return; }
    var p=this.place(text,c); io.set(p.text,p.cursor); io.ok();
    var warn=(r.unknown.length?" · ⚠️ не распознано: "+r.unknown.length:"")+(r.stops.length?" · ⛔ вне рутины: "+r.stops.length:"");
    io.hint("✅ Заключение собрано"+warn,(r.unknown.length||r.stops.length)?6000:2500); },

  /* ---------- панель разбора ---------- */
  infoButton:function(api){ if(document.getElementById("vnk-info-btn")) return;
    var p=document.querySelector(".tools-panel"); if(!p) return;

      return {label:(VNK_SIDE[f.side]||"—")+" · "+d,src:f.src}; }),"#007cba");
    sect("⚠️ Не распознано",r.unknown.map(function(t){ return {label:t,src:t}; }),"#e0a800");
    sect("⛔ Вне рутины (пишите вручную)",r.stops.map(function(t){ return {label:t,src:t}; }),"#dc3545");
    var apply=document.createElement("button"); apply.textContent="Вставить заключение";
    apply.style.cssText="width:100%;padding:8px;margin-top:12px;background:#28a745;color:#fff;border:none;border-radius:4px;cursor:pointer;";
